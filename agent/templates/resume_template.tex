% Custom commands
% \looseness=-1 (layout spec A): each bullet is asked to prefer one fewer
% line, so a lonely short last line ("(27-point gain).") is pulled up when
% the width allows. It is a line-count preference only - text is untouched.
\newcommand{\resumeItem}[1]{
  \item\small{
    \looseness=-1
    {#1 \vspace{-2pt}}
  }
}

% Entry header rows (layout spec B, Nov 11 2026), pdfplumber-measured on the
% run-B render:
%  - leading \vspace -1pt (was -2pt): with the base itemize's rubber
%    itemsep, the bullet-to-next-title gap measures 22.7pt baseline
%    distance (gap 12.7pt) - comfortably above the 6pt target AND robust
%    under page-pressure compression (rubber glue shrinks at most ~3pt,
%    leaving gap >=9pt, far above the 539f7c1d collision floor).
%  - width \textwidth-0.15in (= the outer list indent) puts the tabular's
%    right edge exactly on the section-rule/bullet right edge, so dates
%    right-align with the rules (base rule x1 579.6pt; compact 583.2pt).
%  - trailing \vspace -5.8pt (was -7pt): the tech-stack line sits 13.2pt
%    above its first bullet (gap 4.2pt, well above the 1.5pt floor) while
%    the title->stack gap stays 2.8pt.
\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{\dimexpr\textwidth-0.15in\relax}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-5.8pt}
}
